cmake_minimum_required(VERSION 3.16)
project(atd_sensors)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
# Docker builds the pinned SDK from source. Its CMake package is realsense2;
# the rosdep/system dependency is named librealsense2.
find_package(realsense2 2.56.3 EXACT REQUIRED)
ament_auto_add_executable(atd_realsense_probe src/realsense_probe.cpp)
target_compile_features(atd_realsense_probe PRIVATE cxx_std_20)
target_compile_options(atd_realsense_probe PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atd_realsense_probe ${realsense2_LIBRARY})
ament_python_install_package(atd_sensors)
install(PROGRAMS scripts/d455_setup DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_probe_request test/test_probe_request.cpp)
  target_compile_features(test_probe_request PRIVATE cxx_std_20)
  target_include_directories(test_probe_request PRIVATE include)
  add_test(NAME d455_probe_request COMMAND test_probe_request)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(d455_profile test/test_profile.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(d455_tui test/test_tui.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(d455_launch test/test_launch.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
