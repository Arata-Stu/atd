cmake_minimum_required(VERSION 3.16)
project(atd_rc_input)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
find_package(yaml-cpp REQUIRED)
ament_auto_add_library(atd_rc_input_component SHARED src/rc_input_node.cpp)
target_compile_features(atd_rc_input_component PUBLIC cxx_std_20)
target_compile_options(atd_rc_input_component PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atd_rc_input_component yaml-cpp)
rclcpp_components_register_node(atd_rc_input_component
  PLUGIN "atd_rc_input::RcInputNode" EXECUTABLE atd_rc_input_node)
ament_python_install_package(atd_rc_tools)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_rc_core test/test_core.cpp)
  target_compile_features(test_rc_core PRIVATE cxx_std_20)
  target_include_directories(test_rc_core PRIVATE include)
  add_test(NAME rc_input_core COMMAND test_rc_core)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(rc_profile test/test_profile.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
