cmake_minimum_required(VERSION 3.16)
project(atd_teleop)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
find_package(yaml-cpp REQUIRED)
ament_auto_add_library(atd_teleop_component SHARED src/teleop_node.cpp)
target_compile_features(atd_teleop_component PUBLIC cxx_std_20)
target_compile_options(atd_teleop_component PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atd_teleop_component yaml-cpp)
rclcpp_components_register_node(atd_teleop_component
  PLUGIN "atd_teleop::TeleopNode" EXECUTABLE atd_teleop_node)
ament_python_install_package(atd_teleop_tools)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_teleop test/test_teleop.cpp)
  target_compile_features(test_teleop PRIVATE cxx_std_20)
  target_include_directories(test_teleop PRIVATE include)
  add_test(NAME teleop_core COMMAND test_teleop)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(teleop_profile test/test_profile.py
    APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
