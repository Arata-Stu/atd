cmake_minimum_required(VERSION 3.16)
project(atd_localization)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})

find_package(Threads REQUIRED)
ament_auto_add_library(stereo_bag_input SHARED src/stereo_bag_input.cpp)
target_compile_features(stereo_bag_input PRIVATE cxx_std_17)
target_link_libraries(stereo_bag_input Threads::Threads)
rclcpp_components_register_nodes(stereo_bag_input "atd_localization::StereoBagInput")

find_package(Eigen3 REQUIRED)
ament_auto_add_library(rpm_imu_odometry SHARED src/rpm_imu_odometry.cpp)
target_compile_features(rpm_imu_odometry PRIVATE cxx_std_17)
target_link_libraries(rpm_imu_odometry Eigen3::Eigen)
rclcpp_components_register_nodes(rpm_imu_odometry "atd_localization::RpmImuOdometry")

install(PROGRAMS scripts/compare_rpm_imu scripts/localization_monitor scripts/validate_bag DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(rpm_filter_test test/rpm_filter_test.cpp)
  target_link_libraries(rpm_filter_test Eigen3::Eigen)
  add_test(NAME rpm_filter_test COMMAND rpm_filter_test)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(localization_tests test APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_auto_package()
