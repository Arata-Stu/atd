cmake_minimum_required(VERSION 3.16)
project(ATDOrbslam3Test LANGUAGES CXX)
find_package(ATDOrbslam3 CONFIG REQUIRED)
add_executable(atd_orbslam3_smoke fixed_map.cpp)
target_compile_features(atd_orbslam3_smoke PRIVATE cxx_std_17)
target_link_libraries(atd_orbslam3_smoke PRIVATE ATDOrbslam3::core)
add_executable(atd_orbslam3_sim3 sim3.cpp)
target_compile_features(atd_orbslam3_sim3 PRIVATE cxx_std_17)
target_link_libraries(atd_orbslam3_sim3 PRIVATE ATDOrbslam3::core ATDOrbslam3::DBoW2)
add_executable(atd_orbslam3_imu_frame imu_frame.cpp)
target_compile_features(atd_orbslam3_imu_frame PRIVATE cxx_std_17)
target_link_libraries(atd_orbslam3_imu_frame PRIVATE ATDOrbslam3::core ATDOrbslam3::DBoW2)
enable_testing()
add_test(NAME sim3_ransac COMMAND atd_orbslam3_sim3)
add_test(NAME featureless_stereo_imu COMMAND atd_orbslam3_imu_frame)
