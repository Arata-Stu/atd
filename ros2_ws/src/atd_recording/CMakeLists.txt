cmake_minimum_required(VERSION 3.16)
project(atd_recording)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
ament_auto_add_library(atd_recording_component SHARED
  src/recording_manager.cpp src/native_session.cpp)
target_compile_features(atd_recording_component PUBLIC cxx_std_20)
target_compile_options(atd_recording_component PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atd_recording_component yaml-cpp Threads::Threads
  rosbag2_transport::rosbag2_transport rosbag2_cpp::rosbag2_cpp rosbag2_storage::rosbag2_storage)
rclcpp_components_register_node(atd_recording_component
  PLUGIN "atd_recording::RecordingManager" EXECUTABLE atd_recording_node)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_recording_core test/test_core.cpp)
  target_compile_features(test_recording_core PRIVATE cxx_std_20)
  target_include_directories(test_recording_core PRIVATE include)
  add_test(NAME recording_core COMMAND test_recording_core)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(recording_contract test/test_contract.py)
  ament_add_pytest_test(recording_ros test/test_recording_ros.py TIMEOUT 65)
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
