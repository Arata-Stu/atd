cmake_minimum_required(VERSION 3.16)
project(atd_vehicle_description)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
find_package(yaml-cpp REQUIRED)
ament_auto_add_library(atd_vehicle_description_component SHARED src/vehicle_description_node.cpp)
target_compile_features(atd_vehicle_description_component PUBLIC cxx_std_20)
target_compile_options(atd_vehicle_description_component PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atd_vehicle_description_component yaml-cpp)
rclcpp_components_register_node(atd_vehicle_description_component
  PLUGIN "atd_vehicle_description::VehicleDescriptionNode" EXECUTABLE atd_vehicle_description_node)
ament_python_install_package(atd_vehicle_layout)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(layout_profile test/test_profile.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_layout test/test_layout.cpp)
  target_compile_features(test_layout PRIVATE cxx_std_20)
  target_include_directories(test_layout PRIVATE include)
  add_test(NAME vehicle_layout COMMAND test_layout)
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
