cmake_minimum_required(VERSION 3.16)
project(atd_orbslam3)
find_package(ament_cmake_auto REQUIRED)
option(ATD_REQUIRE_ORBSLAM3 "Fail the configure step if the experimental core is absent" OFF)
set(orb_image_prefix /opt/atd/orbslam3)
# Both development images share the colcon build tree. Detect when switching
# images makes the previously absent core available, even on incremental builds.
file(GLOB orb_image_config CONFIGURE_DEPENDS
  "${orb_image_prefix}/lib/cmake/ATDOrbslam3/ATDOrbslam3Config.cmake")
set(orb_find_mode QUIET)
if(ATD_REQUIRE_ORBSLAM3 OR orb_image_config OR NOT "$ENV{ATD_ORBSLAM3_REV}" STREQUAL "")
  set(orb_find_mode REQUIRED)
endif()
find_package(ATDOrbslam3 ${orb_find_mode} CONFIG PATHS "${orb_image_prefix}")
if(ATDOrbslam3_FOUND)
  ament_auto_find_build_dependencies()
  find_package(OpenCV REQUIRED COMPONENTS core)
  find_package(Eigen3 REQUIRED)
  ament_auto_add_library(orbslam3_component SHARED src/node.cpp src/calibration.cpp)
  target_compile_features(orbslam3_component PUBLIC cxx_std_17)
  get_filename_component(orb_prefix "${ATDOrbslam3_DIR}/../../.." ABSOLUTE)
  set_property(TARGET orbslam3_component APPEND PROPERTY INSTALL_RPATH "${orb_prefix}/lib")
  target_compile_definitions(orbslam3_component PRIVATE
    ATD_ORB_VOCABULARY="${orb_prefix}/share/atd_orbslam3/ORBvoc.txt")
  target_include_directories(orbslam3_component PRIVATE ${OpenCV_INCLUDE_DIRS})
  # ament_auto_add_library uses the plain target_link_libraries signature.
  target_link_libraries(orbslam3_component ATDOrbslam3::core Eigen3::Eigen ${OpenCV_LIBRARIES})
  rclcpp_components_register_node(orbslam3_component
    PLUGIN "atd_orbslam3::OrbslamNode" EXECUTABLE orbslam3_node)
else()
  message(STATUS "ORB-SLAM3 core is absent. Skipping the optional orbslam3_node.")
endif()
if(BUILD_TESTING)
  add_executable(test_orbslam_confirmation test/test_confirmation.cpp)
  target_compile_features(test_orbslam_confirmation PRIVATE cxx_std_17)
  add_test(NAME orbslam_confirmation COMMAND test_orbslam_confirmation)
  if(ATDOrbslam3_FOUND)
    add_executable(test_orbslam_calibration test/test_calibration.cpp src/calibration.cpp)
    target_compile_features(test_orbslam_calibration PRIVATE cxx_std_17)
    target_link_libraries(test_orbslam_calibration PRIVATE
      ${sensor_msgs_TARGETS} Eigen3::Eigen ${OpenCV_LIBRARIES})
    target_include_directories(test_orbslam_calibration PRIVATE ${OpenCV_INCLUDE_DIRS})
    add_test(NAME orbslam_calibration COMMAND test_orbslam_calibration)
  endif()
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
